const VirtualKeyboard = ({ onKeyPress }) => {
  const rows = [
    ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"],
    ["Q", "W", "E", "R", "T", "Y", "U", "I", "O", "P"],
    ["A", "S", "D", "F", "G", "H", "J", "K", "L"],
    ["Z", "X", "C", "V", "B", "N", "M"],
  ];

  return (
    <div>
      {rows.map((row, rowIndex) => (
        <div key={rowIndex} className="flex justify-center mb-1">
          {row.map((key) => (
            <button
              key={key}
              type="button"
              onClick={() => onKeyPress(key)}
              className="w-7 h-8 m-0.5 flex items-center justify-center bg-white text-gray-800 rounded shadow-sm hover:bg-blue-50 border border-gray-200 text-sm"
            >
              {key}
            </button>
          ))}
        </div>
      ))}
      <div className="flex justify-center">
        <button
          type="button"
          onClick={() => onKeyPress("backspace")}
          className="w-16 h-8 m-0.5 flex items-center justify-center bg-white text-gray-800 rounded shadow-sm hover:bg-red-50 border border-gray-200 text-sm"
        >
          ←
        </button>
        <button
          type="button"
          onClick={() => onKeyPress(" ")}
          className="w-32 h-8 m-0.5 flex items-center justify-center bg-white text-gray-800 rounded shadow-sm hover:bg-blue-50 border border-gray-200 text-sm"
        >
          Space
        </button>
      </div>
    </div>
  );
};

const MainScreen = ({ onSubmit, initialVehicleNumber = "", onCancel }) => {
  const [vehicleNumber, setVehicleNumber] = useState(initialVehicleNumber);
  const [showKeyboard, setShowKeyboard] = useState(false);
  const [isEditing, setIsEditing] = useState(!!initialVehicleNumber);

  const handleSubmit = (e) => {
    e.preventDefault();
    if (vehicleNumber.trim()) {
      onSubmit(vehicleNumber.trim());
    }
  };

  const handleClear = () => {
    setVehicleNumber("");
  };

  const toggleKeyboard = () => {
    setShowKeyboard(!showKeyboard);
  };

  const handleKeyPress = (key) => {
    if (key === "backspace") {
      setVehicleNumber((prev) => prev.slice(0, -1));
    } else {
      setVehicleNumber((prev) => prev + key);
    }
  };

  return (
    <div className="w-full max-w-md">
      <div className="bg-white rounded-lg shadow-lg overflow-hidden">
        <div className="bg-blue-700 p-2 text-white flex items-center">
          <button
            onClick={onCancel}
            className="p-1 rounded-full hover:bg-blue-600"
          >
            <ArrowLeft className="h-5 w-5" />
          </button>
          <h2 className="text-xl font-bold text-center flex-grow">
            {isEditing ? "Edit Vehicle Number" : "ANPR Detection"}
          </h2>
        </div>

        <form onSubmit={handleSubmit} className="p-3">
          {/* Vehicle Number Display/Edit */}
          <div className="mb-3">
            <div className="bg-blue-50 border-2 border-blue-200 rounded-lg p-3">
              <div className="flex items-center justify-between mb-1">
                <div className="flex items-center">
                  <Car className="h-4 w-4 text-blue-600 mr-1" />
                  <span className="text-blue-800 font-medium text-sm">
                    Detected Vehicle Number
                  </span>
                </div>
                {isEditing && (
                  <div className="flex items-center text-blue-600 text-xs">
                    <Edit className="h-3 w-3 mr-1" />
                    <span>Editing</span>
                  </div>
                )}
              </div>

              <div className="relative">
                <input
                  type="text"
                  value={vehicleNumber}
                  onChange={(e) => {
                    setVehicleNumber(e.target.value);
                  }}
                  placeholder="Enter Vehicle Number"
                  className="w-full text-2xl font-bold text-center py-3 bg-white rounded-md border border-blue-200 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                />

                {vehicleNumber && (
                  <button
                    type="button"
                    onClick={handleClear}
                    className="absolute right-2 top-1/2 transform -translate-y-1/2 p-1 bg-gray-100 rounded-full hover:bg-gray-200"
                  >
                    <X className="h-4 w-4 text-gray-600" />
                  </button>
                )}
              </div>
            </div>
          </div>

          <button
            type="button"
            onClick={toggleKeyboard}
            className="mb-3 flex items-center justify-center py-2 px-3 bg-gray-50 text-gray-700 rounded-md hover:bg-gray-100 border border-gray-200 text-sm w-full"
          >
            <Keyboard className="h-4 w-4 mr-1" />
            <span>
              {showKeyboard ? "Hide Keyboard" : "Show Virtual Keyboard"}
            </span>
          </button>

          {showKeyboard && (
            <div className="mb-3 bg-gray-50 p-2 rounded-md border border-gray-200">
              <VirtualKeyboard onKeyPress={handleKeyPress} />
            </div>
          )}

          <div className="flex space-x-3">
            <button
              type="button"
              onClick={onCancel}
              className="flex-1 py-2 text-sm font-medium rounded-md bg-gray-100 text-gray-700 hover:bg-gray-200 border border-gray-300"
            >
              Cancel
            </button>

            <button
              type="submit"
              disabled={!vehicleNumber.trim()}
              className={`flex-1 py-2 text-sm font-medium rounded-md ${
                vehicleNumber.trim()
                  ? "bg-blue-600 text-white hover:bg-blue-700"
                  : "bg-gray-300 text-gray-500 cursor-not-allowed"
              }`}
            >
              Continue
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};

const PrintPage = ({ onPrintAgain, onHome }) => {
  const [isPrinting, setIsPrinting] = useState(false);
  const [printSuccess, setPrintSuccess] = useState(false);
  const [apiData, setApiData] = useState(null);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchData = async () => {
      setIsPrinting(true);
      try {
        const response = await fetch("http://127.0.0.1:5000/receive_data");
        console.log("Response:", response);
        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }
        const jsonData = await response.json();
        setApiData(jsonData.data);
        setIsPrinting(false);
        setPrintSuccess(true);
        setTimeout(() => setPrintSuccess(false), 2000);
      } catch (e) {
        setError(e.message);
        setIsPrinting(false);
      }
    };

    fetchData();
  }, []);

  if (error) {
    return (
      <div className="w-full max-w-md">
        <div className="bg-white rounded-lg shadow-md p-6 text-center">
          <h2 className="text-xl font-bold text-red-600 mb-4">Error</h2>

            onClick={onHome}
            className="mt-4 py-2 px-4 bg-gray-100 text-gray-700 rounded-md hover:bg-gray-200 border border-gray-300 text-sm"
          >
            Return Home
          </button>
        </div>
      </div>
    );
  }

      <div className="w-full max-w-md">
        <div className="bg-white rounded-lg shadow-md p-6 text-center">

            Fetching Pass Details...

const CurrentTime = () => {
  const [time, setTime] = React.useState(new Date());

  React.useEffect(() => {
    const timer = setInterval(() => {
      setTime(new Date());
    }, 1000);
    return () => clearInterval(timer);
  }, []);

  const formattedDate = time.toLocaleDateString("en-US", {
    weekday: "short",
    month: "short",
    day: "numeric",
  });

  return (
    <div className="text-right">
      <span className="text-sm font-medium">
        {time.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
      </span>
      <div className="text-xs text-blue-100">{formattedDate}</div>
    </div>
  );
};

const HospitalKioskApp = () => {
  const [screen, setScreen] = useState("video");
  const [vehicleNumber, setVehicleNumber] = useState("");
  const [lastActivity, setLastActivity] = useState(Date.now());

  // Reset to video screen after inactivity, but only on video screen
  useEffect(() => {
    const inactivityTimer = setInterval(() => {
      if (Date.now() - lastActivity > 15000 && screen === "video") {
        setScreen("video");
        setVehicleNumber("");
      }
    }, 1000);

    return () => clearInterval(inactivityTimer);
  }, [lastActivity, screen]);

  // Update last activity timestamp
  const updateActivity = () => {
    setLastActivity(Date.now());
  };

  // Handle user interactions
  const handleUserInteraction = () => {
    updateActivity();
  };

  useEffect(() => {
    window.addEventListener("click", handleUserInteraction);
    window.addEventListener("touchstart", handleUserInteraction);
    window.addEventListener("keydown", handleUserInteraction);

    return () => {
      window.removeEventListener("click", handleUserInteraction);
      window.removeEventListener("touchstart", handleUserInteraction);
      window.removeEventListener("keydown", handleUserInteraction);
    };
  }, []);

  const handleVehicleDetected = (number) => {
    setVehicleNumber(number);
    // Go directly to confirmation page instead of main screen
    setScreen("confirmation");
  };

  const handleVehicleNumberSubmit = (number) => {
    setVehicleNumber(number);
    setScreen("confirmation");
  };

  const handleConfirm = () => {
    setScreen("print");
  };

  const handlePrintAgain = () => {
    setScreen("print"); // Re-fetching data will happen in PrintPage
  };

  const handleReturnHome = () => {
    setScreen("video");
    setVehicleNumber("");
  };

  const handleEditNumber = () => {
    setScreen("main");
  };

  return (
    <div
      className="min-h-screen bg-gray-100 flex flex-col"
      onClick={updateActivity}
    >
      {screen === "video" ? (
        <VideoScreen onVehicleDetected={handleVehicleDetected} />
      ) : (
        <>
          <Header />

          <div className="flex-grow flex items-center justify-center p-2">
            {screen === "main" && (
              <MainScreen
                onSubmit={handleVehicleNumberSubmit}
                initialVehicleNumber={vehicleNumber}
                onCancel={handleReturnHome}
              />
            )}

            {screen === "confirmation" && (
              <ConfirmationPage
                vehicleNumber={vehicleNumber}
                onConfirm={handleConfirm}
                onCancel={handleReturnHome}
                onEdit={handleEditNumber}
              />
            )}

            {screen === "print" && (
              <PrintPage
                onPrintAgain={handlePrintAgain}
                onHome={handleReturnHome}
              />
            )}
          </div>

          <footer className="bg-white border-t border-gray-200 py-1 text-center text-gray-500 text-xs">
            <p>© 2023 Pro Parking</p>
          </footer>
        </>
      )}
    </div>
  );
};

export default HospitalKioskApp;
